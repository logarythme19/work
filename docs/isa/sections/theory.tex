\section{LQI synthesis in weight space and its exact 2-DOF PIDF realization}\label{sec:theory}

\subsection{Integral augmentation and Bryson weights}
Let $z(t)=\int_0^t(r-v_o)\,d\tau$. Around the synthesis equilibrium ($V_o=\SI{24}{V}$)
the augmented plant is
\begin{equation}
\dot x_a=A_ax_a+B_a\delta d,\quad
A_a=\begin{bmatrix}A&0\\-C_y&0\end{bmatrix},\;
B_a=\begin{bmatrix}B_d\\0\end{bmatrix},\;
x_a=\begin{bmatrix}\delta x\\z\end{bmatrix},
\label{eq:aug}
\end{equation}
and the LQI law $\delta d=-Kx_a$, $K=[K_{ix}\;K_{vx}\;K_z]$, minimizes
$\int_0^\infty(x_a^\top Qx_a+R_d\delta d^2)\,dt$ with
\begin{equation}
A_a^\top P+PA_a-PB_aR_d^{-1}B_a^\top P+Q=0,\qquad K=R_d^{-1}B_a^\top P .
\label{eq:care}
\end{equation}
Bryson's rule assigns $Q=\mathrm{diag}(\Delta_i^{-2},\Delta_v^{-2},(\Delta_v/\omega_B)^{-2})$
and $R_d=\Delta_d^{-2}$: $\Delta_i$ and $\Delta_v$ are admissible current and
voltage excursions and the integral state, in \si{V.s}, is scaled by a target
bandwidth $\omega_B$. The duty excursion is fixed at its physical value
$\Delta_d=(d_{\max}-d_{\min})/2=0.48$, which removes the scale redundancy
$(Q,R_d)\to(\alpha Q,\alpha R_d)$ of \eqref{eq:care}. The decision vector of the
co-design is
\begin{equation}
\xi=[\log_{10}\Delta_i,\;\log_{10}\Delta_v,\;\log_{10}\omega_B,\;\log_{10}K_b]^\top ,
\label{eq:xi}
\end{equation}
where $K_b$ is the back-calculation rate of the anti-windup, which lies outside
the linear theory and is searched alongside the weights. The normalization
record $\xi_0$ uses $\Delta_i=\SI{15}{A}$, $\Delta_v=\SI{46}{V}$ (the largest
commanded output) and $\omega_B=K_b=2\pi\cdot\SI{1500}{rad/s}$.

\subsection{Frequency-domain realization}
For $i_p=0$ the output network gives, in the Laplace domain,
$v_C=v_o/(1+r_CCs)$ and $i_L=v_o/R+Csv_o/(1+r_CCs)$. Substituting into the state
part of the LQI law,
\begin{equation}
\frac{K_{ix}i_L+K_{vx}v_C}{v_o}=\frac{K_{ix}}{R}+\frac{K_{ix}Cs+K_{vx}}{1+r_CCs}
=K_p+K_d\frac{N_Cs}{s+N_C},
\label{eq:split}
\end{equation}
which exposes one constant and one filtered derivative term.

\begin{proposition}[Nominal realization]\label{prop:map}
The output-error compensator generated by the full-state LQI law is the 2-DOF
PIDF
\begin{equation}
d=d_{ff}(r)+K_p(br-v_o)+\frac{K_i}{s}(r-v_o)+K_d\frac{N_Cs}{s+N_C}(cr-v_o),
\label{eq:pidf}
\end{equation}
with
\begin{equation}
\begin{aligned}
K_p&=K_{vx}+\frac{K_{ix}}{R},\quad K_i=-K_z,\quad K_d=C(K_{ix}-r_CK_{vx}),\\
N&=N_C=\frac{1}{r_CC},\quad b=1,\quad c=0 .
\end{aligned}
\label{eq:map}
\end{equation}
\end{proposition}
\begin{proof}
Split the right-hand side of \eqref{eq:split} into its zero-frequency value
$K_{vx}+K_{ix}/R$ and the remainder $C(K_{ix}-r_CK_{vx})N_Cs/(s+N_C)$. The
equilibrium map $x_{eq}(r)=[r/R\;\;r]^\top$ contributes $+(K_{ix}/R+K_{vx})r=K_pr$,
hence $b=1$; the state-feedback law has no derivative of the reference, hence
$c=0$; the integral state contributes $-K_zz$, hence $K_i=-K_z$.
\end{proof}

Because $N_C=\SI{2e5}{rad/s}<\pi/T_s=\SI{3.14e5}{rad/s}$, the exact filter pole is
implementable at the adopted sampling period, and no projected surrogate is
needed: every candidate of the search is realized with its own exact pole.

\subsection{Structural (trajectory-level) equivalence}
Proposition~\ref{prop:map} is a statement about transfer functions, i.e.\ about
the linear nominal plant with consistent initialization. The next result shows
that the realization is in fact exact along every trajectory of the
\emph{nonlinear} converter, because it only uses the output network, never the
inductor branch. Write the PIDF of \eqref{eq:pidf} in state form,
\begin{equation}
\begin{aligned}
\tilde d&=d_{ff}(r)+K_p(br-v_o)+x_I+K_dN(cr-v_o-x_F),\qquad d=\mathrm{sat}(\tilde d),\\
\dot x_I&=K_i(r-v_o)+K_b(d-\tilde d),\qquad \dot x_F=N(cr-v_o-x_F),
\end{aligned}
\label{eq:pidfss}
\end{equation}
and the LQI with the same feedforward and anti-windup as
\begin{equation}
\begin{aligned}
\tilde d&=d_{ff}(r)-K_{ix}\big(i_L-\tfrac{r}{R}\big)-K_{vx}(v_C-r)-K_zz,\\
\dot z&=(r-v_o)+\tfrac{K_b}{|K_z|}(d-\tilde d).
\end{aligned}
\label{eq:lqiss}
\end{equation}

\begin{proposition}[Structural equivalence]\label{prop:struct}
Let $r$ be piecewise constant, let the PIDF use \eqref{eq:map} with $c=0$,
$N=N_C$ computed from the true $(r_C,C)$, and let $x_F(0)=-v_C(0)$ and
$x_I(0)=-K_zz(0)$. Then, along every trajectory of the converter---averaged or
switched, in continuous or discontinuous conduction, with or without active
saturation and anti-windup, and for any values of $L$, $r_L$, $R_{on}$, $r_d$,
$V_f$ and $V_{in}$:
\begin{enumerate}
\item[(i)] $x_F(t)=-v_C(t)$ for all $t\ge0$;
\item[(ii)] the two unsaturated commands satisfy
\begin{equation}
\tilde d_{LQI}(t)-\tilde d_{PIDF}(t)=-K_{ix}\Big(i_o(t)-\frac{v_o(t)}{R}\Big),
\label{eq:gap}
\end{equation}
where $i_o=i_L-i_C$ is the actual output-branch current and $R$ the nominal load;
\item[(iii)] if $i_o\equiv v_o/R$ (no load disturbance, nominal load), the
two closed loops generate identical trajectories.
\end{enumerate}
\end{proposition}
\begin{proof}
(i) The capacitor branch gives $v_o=v_C+r_CC\dot v_C$ exactly, whatever flows
into the output node, so $\dot v_C=N_C(v_o-v_C)$. Let $\varepsilon=x_F+v_C$.
Between reference steps, with $c=0$,
$\dot\varepsilon=N_C(-v_o-x_F)+N_C(v_o-v_C)=-N_C\varepsilon$, and $\varepsilon$
is continuous across steps because $c=0$. Hence $\varepsilon(0)=0$ implies
$\varepsilon\equiv0$.
(ii) With (i), the derivative term of \eqref{eq:pidfss} is
$K_dN_C(v_C-v_o)=-K_dN_Cr_Ci_C=-(K_{ix}-r_CK_{vx})i_C$. In \eqref{eq:lqiss}
substitute $i_L=i_o+i_C$ and $v_C=v_o-r_Ci_C$:
$-K_{ix}(i_L-r/R)-K_{vx}(v_C-r)=K_p(r-v_o)-(K_{ix}-r_CK_{vx})i_C-K_{ix}(i_o-v_o/R)$.
With $b=1$ and $x_I=-K_zz$ the difference of the two commands is
\eqref{eq:gap}.
(iii) If the right-hand side of \eqref{eq:gap} vanishes, both laws issue the same
$\tilde d$ and $d$ at every instant. Then
$\frac{d}{dt}(x_I+K_zz)=K_i(r-v_o)+K_b(d-\tilde d)-|K_z|(r-v_o)-K_b(d-\tilde d)=0$
because $K_z<0$, so $x_I=-K_zz$ is preserved, and the two closed-loop systems
coincide; uniqueness of solutions gives identical trajectories.
\end{proof}

Three practical consequences follow. First, the PIDF does not approximate the
LQI on the nonlinear converter; it \emph{is} the LQI whenever the load current
is the nominal one, and the saturation, anti-windup and conduction-mode effects
that invalidate small-signal arguments do not break it. Second, the price of
removing the current sensor is stated exactly by \eqref{eq:gap}: the full-state
law contains an instantaneous proportional feedback $-K_{ix}(i_o-v_o/R)$ on the
unmodelled output current, which the PIDF must recover through its integral
action. Third, the equivalence is lost by exactly three mechanisms: a load
disturbance or load mismatch (through \eqref{eq:gap}), a mismatch of $(r_C,C)$
(through $N\neq N_C$ in (i)), and sampling, because the discrete filter
reconstructs $v_C$ only between samples while the sampled LQI reads $i_L$ with
its switching ripple. Section~\ref{sec:res:equiv} verifies each case.

\subsection{Guarantees and their scope}
\paragraph{LQR margins} For the full-state augmented loop
$L_K(s)=K(sI-A_a)^{-1}B_a$ the return-difference identity gives
$|1+L_K(j\omega)|\ge1$, hence the gain margin $(1/2,\infty)$ and a phase margin of
at least \SI{60}{\degree} \citep{Safonov1977,Anderson1990}. By
Proposition~\ref{prop:struct}(iii) these margins are those of the PIDF loop at the
input of the nominal linear plant; they do not survive sampling or an observer
\citep{Doyle1978}.

\paragraph{Operating-polytope certificate} The augmented closed-loop matrix is
affine in $(\rho_0(d_0),E_d(I_{L,0}))$. Over $v_o\in[1,46]$\,V and $i_p\in[0,0.5]$\,A
the rectangle $d_0\in[\Dmin,\Dmax]$, $I_{L,0}\in[0.1,5.1]$\,A has four vertices
$A_{cl,k}$. A common $P\succ0$ with
\begin{equation}
A_{cl,k}P+PA_{cl,k}^\top+2\alpha P\prec0,\quad
\begin{bmatrix}\sin\theta(M_k+M_k^\top)&\cos\theta(M_k-M_k^\top)\\
\cos\theta(M_k^\top-M_k)&\sin\theta(M_k+M_k^\top)\end{bmatrix}\prec0,
\label{eq:lmi}
\end{equation}
$M_k=A_{cl,k}P$, $\zeta=\cos\theta$, places every closed-loop matrix of the convex
hull in $\{\Re s<-\alpha\}\cap\{\zeta(s)>\zeta\}$ \citep{Chilali1996}. Bisection
gives the boundary values; a \SI{1}{\percent} safety factor is applied and the
recovered $P$ is re-checked for strictness.

\paragraph{Discrete implementation}
All controllers run at $T_s$ with a one-sample computational delay, total duty
saturation and forward-Euler integral and back-calculation updates.
Proposition~\ref{prop:struct}(i) says that the derivative-filter state is an
observer of the capacitor voltage, $x_F=cr-v_C$, and it should be discretized as
one. The usual zero-order-hold discretization,
$x_F[k{+}1]=\phi x_F[k]+(1-\phi)w[k]$ with $w=cr-v_o$ and $\phi=e^{-NT_s}$, uses at
sample $k$ an estimate built from $v_o[k{-}1]$ only; during a \SI{45}{V} slew
$v_o$ moves by about \SI{2.7}{V} per sample and $K_dN\approx0.7$ turns this lag
into a duty error of order one. We therefore use the triangle-hold (first-order
hold) update, exact when $v_o$ is piecewise linear between samples,
\begin{equation}
x_F[k]=\phi\,x_F[k{-}1]+(1-\phi)\,w[k{-}1]+\gamma\big(w[k]-w[k{-}1]\big),\quad
\gamma=1-\frac{1-\phi}{NT_s},
\label{eq:foh}
\end{equation}
evaluated with the current sample before the command
$\tilde d[k]=d_{ff}+K_p(br[k]-v_o[k])+x_I[k]+K_dN(w[k]-x_F[k])$ is formed, and
initialized with $x_F[0]=cr[0]-v_C(0)$. Section~\ref{sec:res:equiv} shows that
this single choice decides whether the sampled PIDF remains equivalent to the
sampled LQI.

\subsection{Numerical instantiation}\label{sec:instantiation}
At the \SI{24}{V} point, \eqref{eq:ss} evaluates to
\begin{equation}
\begin{gathered}
A=\begin{bmatrix}\Aa&\Ab\\\Ac&\Ad\end{bmatrix}\si{s^{-1}},\\
B_d=\begin{bmatrix}\Bd\\0\end{bmatrix},\quad
C_y=\begin{bmatrix}\Cya&\Cyb\end{bmatrix}.
\end{gathered}
\end{equation}
For $\xi_0$, $Q=\mathrm{diag}(\Qa,\Qb,\Qc)$, $R_d=\Rd$ and \eqref{eq:care} gives
$K=[\Ka\;\;\Kb\;\;\Kc]$ with a relative Riccati residual of \CAREres{} and
closed-loop poles \Poles{}. The realization \eqref{eq:map} gives
\begin{equation}
\begin{aligned}
\theta_0&=[K_p,K_i,K_d,N,b,c,K_b]\\
&=[\ThA,\ \ThB,\ \ThC,\ 2\times10^{5},\ 1,\ 0,\ \ThG],
\end{aligned}
\end{equation}
with a zero coefficient residual and a maximum relative frequency-domain error of
\FreqErr{} over $[10^{-1},10^{8}]$\,rad/s. The minimum return difference is
\RDmin{} and the phase margin of the full-state loop is \PMdeg. The
operating-polytope certificate at $\xi_0$ is
$\alpha_{cert}=\SI{\AlphaCert}{rad/s}$ and $\zeta_{cert}=\ZetaCert$
($\lambda_{\min}(P)>0$ and strictly negative LMI eigenvalues were re-verified on
the recovered matrices).
